<?php
$siteName    = app_setting('website_name', 'AdminHub');
$currentPage = basename($_SERVER['SCRIPT_NAME']);
$customerId  = (int) ($_SESSION['customer_id'] ?? 0);
$customerName = $_SESSION['customer_name'] ?? 'Customer';
$unreadCount = 0;
if ($customerId > 0) {
    try {
        $unreadCount = unread_customer_notification_count($database, $customerId);
    } catch (\Throwable $e) { $unreadCount = 0; }
}

$navItems = [

    ['href' => 'notifications.php', 'label' => 'Notifications',  'icon' => 'bi-bell'],

];

<aside class="customer-sidebar" id="customerSidebar">

    <div class="d-flex flex-column flex-grow-1 overflow-y-auto px-2 py-3">
        <div class="nav-section-label text-muted small text-uppercase px-3 mb-2 fw-semibold" style="font-size: 0.75rem;">Menu</div>
        <nav class="nav flex-column gap-1">
            <?php foreach ($navItems as $item): ?>
                <?php $isActive = $currentPage === $item['href']; ?>
                <a href="<?php echo e($item['href']); ?>" class="nav-link d-flex align-items-center px-3 py-2 rounded text-white <?php echo $isActive ? 'bg-primary active' : 'opacity-75'; ?>">
                    <span class="nav-icon me-3"><i class="bi <?php echo e($item['icon']); ?> fs-5"></i></span>
                    <span class="flex-grow-1"><?php echo e($item['label']); ?></span>
                    
                    <?php if ($item['href'] === 'notifications.php' && $unreadCount > 0): ?>
                        <span class="badge rounded-pill bg-danger" style="font-size:0.65rem;">
                            <?php echo $unreadCount > 99 ? '99+' : $unreadCount; ?>
                        </span>
                    <?php endif; ?>
                    
                    <?php if ($item['href'] === 'cart.php'): ?>
                        <?php
                        $cartCount = 0;
                        try {
                            $cStmt = $database->prepare("SELECT COALESCE(SUM(quantity),0) FROM cart_items WHERE customer_id = :cid");
                            $cStmt->execute(['cid' => $customerId]);
                            $cartCount = (int) $cStmt->fetchColumn();
                        } catch (\Throwable $e) {}
                        if ($cartCount > 0):
                        ?>
                            <span class="badge rounded-pill bg-info text-dark" style="font-size:0.65rem;">
                                <?php echo $cartCount; ?>
                            </span>
                        <?php endif; ?>

        </nav>

<script>
function toggleSidebar() {
    const sidebar = document.getElementById('customerSidebar');
    const overlay = document.getElementById('sidebarOverlay');
    const isOpen  = sidebar.classList.toggle('show');
    overlay.style.display = isOpen ? 'block' : 'none';
}
</script>
