<?php
require_once __DIR__ . '/../config/init.php';
require_admin();

$pageTitle     = 'Orders';
$orderStatuses = ['Pending', 'Confirmed', 'Processing', 'Shipped', 'Out for Delivery', 'Delivered', 'Cancelled'];
$shippingStatuses = ['Processing', 'Packed', 'Shipped', 'Out for Delivery', 'Delivered', 'Returned'];

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!verify_csrf_token($_POST['csrf_token'] ?? null)) {
        flash('danger', 'Security token expired. Please try again.');
        redirect_to('orders.php');
    }

    $action = $_POST['action'] ?? '';

    try {
        // ── Approve Order ─────────────────────────────────────────────────────
        if ($action === 'approve_order') {
            $orderId     = (int) ($_POST['order_id'] ?? 0);
            $trackingUrl = trim($_POST['tracking_url'] ?? '');

            if ($orderId <= 0) {
                flash('danger', 'Invalid order selected.');
                redirect_to('orders.php');
            }

            $stmt = $database->prepare(
                "SELECT o.id, o.txn_id, o.customer_id, o.order_status, o.customer_name
                 FROM orders o WHERE o.id = :id LIMIT 1"
            );
            $stmt->execute(['id' => $orderId]);
            $order = $stmt->fetch();

            if (!$order) {
                flash('danger', 'Order not found.');
                redirect_to('orders.php');
            }

            if ($order['order_status'] !== 'Pending') {
                flash('warning', 'Only Pending orders can be approved.');
                redirect_to('orders.php');
            }

            $adminId = (int) ($_SESSION['admin_id'] ?? 0);

            $database->prepare(
                "UPDATE orders
                 SET order_status = 'Confirmed',
                     approved_by  = :admin_id,
                     approved_at  = NOW(),
                     tracking_url = :tracking_url
                 WHERE id = :id"
            )->execute([
                'admin_id'     => $adminId,
                'tracking_url' => $trackingUrl !== '' ? $trackingUrl : null,
                'id'           => $orderId,
            ]);

            // Notify customer if this was a customer-portal order
            if (!empty($order['customer_id'])) {
                $trackingMsg = $trackingUrl !== ''
                    ? " You can track your shipment at: {$trackingUrl}"
                    : '';
                notify_customer(
                    $database,
                    (int) $order['customer_id'],
                    'order_approved',
                    "Great news! Your order {$order['txn_id']} has been approved and is being prepared.{$trackingMsg}",
                    ['order_id' => $orderId, 'txn_id' => $order['txn_id'], 'tracking_url' => $trackingUrl]
                );
            }

            flash('success', "Order {$order['txn_id']} approved successfully.");

        // ── Update Order Status ───────────────────────────────────────────────
        } elseif ($action === 'update_status') {
            $orderId        = (int) ($_POST['order_id'] ?? 0);
            $orderStatus    = $_POST['order_status'] ?? 'Pending';
            $shippingStatus = $_POST['shipping_status'] ?? 'Processing';
            $trackingUrl    = trim($_POST['tracking_url'] ?? '');

            if ($orderId <= 0
                || !in_array($orderStatus, $orderStatuses, true)
                || !in_array($shippingStatus, $shippingStatuses, true)
            ) {
                flash('danger', 'Invalid order status update.');
                redirect_to('orders.php');
            }

            // Fetch order to notify customer
            $stmt = $database->prepare("SELECT customer_id, txn_id FROM orders WHERE id = :id LIMIT 1");
            $stmt->execute(['id' => $orderId]);
            $order = $stmt->fetch();

            $database->prepare(
                "UPDATE orders
                 SET order_status    = :order_status,
                     shipping_status = :shipping_status,
                     tracking_url    = COALESCE(NULLIF(:tracking_url, ''), tracking_url)
                 WHERE id = :id"
            )->execute([
                'order_status'    => $orderStatus,
                'shipping_status' => $shippingStatus,
                'tracking_url'    => $trackingUrl,
                'id'              => $orderId,
            ]);

            // Notify customer of status change
            if ($order && !empty($order['customer_id'])) {
                $notifMessages = [
                    'Processing'      => "Your order {$order['txn_id']} is now being processed.",
                    'Shipped'         => "Your order {$order['txn_id']} has been shipped!",
                    'Out for Delivery'=> "Your order {$order['txn_id']} is out for delivery today!",
                    'Delivered'       => "Your order {$order['txn_id']} has been delivered. Enjoy!",
                    'Cancelled'       => "Your order {$order['txn_id']} has been cancelled. Contact support for details.",
                ];

                if (isset($notifMessages[$orderStatus])) {
                    notify_customer(
                        $database,
                        (int) $order['customer_id'],
                        'order_status_' . strtolower(str_replace(' ', '_', $orderStatus)),
                        $notifMessages[$orderStatus],
                        ['order_id' => $orderId, 'txn_id' => $order['txn_id'], 'status' => $orderStatus]
                    );
                }
            }

            flash('success', 'Order status updated.');

        // ── Delete Order ──────────────────────────────────────────────────────
        } elseif ($action === 'delete') {
            if (!is_primary_admin() && !is_secondary_admin()) {
                flash('danger', 'You do not have permission to delete orders.');
                redirect_to('orders.php');
            }

            $orderId = (int) ($_POST['order_id'] ?? 0);
            if ($orderId <= 0) {
                flash('danger', 'Invalid order selected.');
                redirect_to('orders.php');
            }

            $database->beginTransaction();
            $database->prepare('DELETE FROM order_items WHERE order_id = :order_id')
                ->execute(['order_id' => $orderId]);
            $database->prepare('DELETE FROM orders WHERE id = :id')
                ->execute(['id' => $orderId]);
            $database->commit();
            flash('success', 'Order deleted.');
        }

    } catch (PDOException $e) {
        if ($database->inTransaction()) {
            $database->rollBack();
        }
        flash('danger', 'Order operation failed. Please try again.');
    }

    redirect_to('orders.php');
}

$orders    = [];

try {
    $stmt = $database->prepare(
        "SELECT
            o.id,
            o.txn_id,
            o.customer_id,
            o.customer_name,
            o.total_amount,
            o.order_status,
            o.shipping_status,
            o.tracking_url,
            o.approved_at,
            o.created_at,
            GROUP_CONCAT(CONCAT(COALESCE(oi.product_name, COALESCE(p.name, 'Deleted Product')), ' x', oi.quantity) ORDER BY oi.id SEPARATOR ', ') AS items,
            a.fullname AS approved_by_name
         FROM orders o
         LEFT JOIN order_items oi ON oi.order_id = o.id
         LEFT JOIN products p ON p.id = oi.product_id
         LEFT JOIN admins a ON a.id = o.approved_by
         GROUP BY o.id, o.txn_id, o.customer_id, o.customer_name, o.total_amount,
                  o.order_status, o.shipping_status, o.tracking_url, o.approved_at, o.created_at, a.fullname
         ORDER BY o.created_at DESC, o.id DESC"
    );
    $stmt->execute();
    $orders = $stmt->fetchAll();
